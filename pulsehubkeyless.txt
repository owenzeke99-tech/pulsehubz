-- Function to run the main script
local function loadMainScript()
    loadstring(game:HttpGet(scriptUrl))()
end

-- Check Auto-Save Key (24 Hours Duration)
if readfile and isfile and isfile(saveFileName) then
    local success, data = pcall(function()
        return game:GetService("HttpService"):JSONDecode(readfile(saveFileName))
    end)
    
    if success and data and data.Key == correctKey and data.Expire then
        if os.time() < data.Expire then
            -- Key is still valid, run script directly!
            loadMainScript()
            return
        else
            -- Expired, delete file
            if delfile then delfile(saveFileName) end
        end
    end
end

-- Smooth Shake Animation
local function shakeFrame()
    local originalPos = UDim2.new(0.5, -210, 0.5, -142)
    local shakeDistance = 7
    local tweenInfo = TweenInfo.new(0.04, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut, 0, true)

    task.spawn(function()
        for _ = 1, 3 do
            local leftTween = TweenService:Create(mainFrame, tweenInfo, {Position = UDim2.new(0.5, -210 - shakeDistance, 0.5, -142)})
            leftTween:Play()
            leftTween.Completed:Wait()

            local rightTween = TweenService:Create(mainFrame, tweenInfo, {Position = UDim2.new(0.5, -210 + shakeDistance, 0.5, -142)})
            rightTween:Play()
            rightTween.Completed:Wait()
        end
        mainFrame.Position = originalPos
    end)
end

-- Modern Status Notification Handler
local currentTask = nil
local function showNotification(text, textColor, bgColor, strokeColor)
    statusLabel.Text = text
    statusLabel.TextColor3 = textColor
    statusBadge.BackgroundColor3 = bgColor
    badgeStroke.Color = strokeColor

    TweenService:Create(statusBadge, TweenInfo.new(0.25), {BackgroundTransparency = 0}):Play()
    TweenService:Create(badgeStroke, TweenInfo.new(0.25), {Transparency = 0.5}):Play()
    TweenService:Create(statusLabel, TweenInfo.new(0.25), {TextTransparency = 0}):Play()

    if currentTask then task.cancel(currentTask) end
    currentTask = task.delay(2.5, function()
        TweenService:Create(statusBadge, TweenInfo.new(0.3), {BackgroundTransparency = 1}):Play()
        TweenService:Create(badgeStroke, TweenInfo.new(0.3), {Transparency = 1}):Play()
        TweenService:Create(statusLabel, TweenInfo.new(0.3), {TextTransparency = 1}):Play()
    end)
end

-- Key Verification Action
local isVerifying = false

local function verifyKey()
    if isVerifying then return end
    isVerifying = true

    if keyInput.Text == correctKey then
        -- Save Key Data for 24 Hours
        if writefile then
            pcall(function()
                local saveData = {
                    Key = correctKey,
                    Expire = os.time() + (24 * 60 * 60) -- 24 Hours in seconds
                }
                writefile(saveFileName, game:GetService("HttpService"):JSONEncode(saveData))
            end)
        end

        showNotification("KEY VERIFIED! LOADING SCRIPT...", COLORS.SuccessGreen, COLORS.SuccessBg, COLORS.SuccessGreen)
        TweenService:Create(inputStroke, TweenInfo.new(0.3), {Color = COLORS.SuccessGreen, Transparency = 0}):Play()
        
        task.wait(1)
        screenGui:Destroy()
        
        -- Load Original Script
        loadMainScript()
    else
        -- Clear Input Text
        keyInput.Text = ""
        
        -- Show Error & Shake
        showNotification("INVALID KEY! PLEASE TRY AGAIN", COLORS.ErrorRed, COLORS.ErrorBg, COLORS.ErrorRed)
        TweenService:Create(inputStroke, TweenInfo.new(0.2), {Color = COLORS.ErrorRed, Transparency = 0}):Play()
        shakeFrame()
        
        task.wait(1.8)
        TweenService:Create(inputStroke, TweenInfo.new(0.3), {Color = COLORS.IceBlue, Transparency = 0.8}):Play()
        
        isVerifying = false
    end
end

checkKeyBtn.MouseButton1Click:Connect(verifyKey)
